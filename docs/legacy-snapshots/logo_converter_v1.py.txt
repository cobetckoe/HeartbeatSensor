# -*- coding: utf-8 -*-
"""
logo_converter.py - 把你的 LOGO 图片转成固件内置点阵

用法：
  1) 把图片放到 utils/ 目录下，命名为 logo.png（或 logo.jpg / logo.bmp）
  2) 运行: python utils/logo_converter.py
  3) 生成 include/LogoImage.h，重新编译烧录即可

图片会自动缩放到 160x128（ST7735 屏幕尺寸）。
依赖: pip install Pillow
"""

import os
import sys

try:
    from PIL import Image
except ImportError:
    print("缺少 Pillow 库，请先安装: pip install Pillow")
    sys.exit(1)

SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(SCRIPT_DIR)
OUT_PATH = os.path.join(ROOT, "include", "LogoImage.h")

SCREEN_W, SCREEN_H = 160, 128
CANDIDATES = ["logo.png", "logo.jpg", "logo.jpeg", "logo.bmp"]


def find_logo():
    for name in CANDIDATES:
        p = os.path.join(SCRIPT_DIR, name)
        if os.path.isfile(p):
            return p
    return None


def to_rgb565(img, max_w=SCREEN_W, max_h=SCREEN_H):
    """等比缩放到不超过屏幕尺寸，再转 RGB565 像素列表。"""
    w, h = img.size
    scale = min(max_w / w, max_h / h, 1.0)
    if scale < 1.0:
        img = img.resize((max(1, int(w * scale)), max(1, int(h * scale))),
                         Image.LANCZOS)
    img = img.convert("RGB")
    px = []
    for r, g, b in img.getdata():
        r5 = (r >> 3) & 0x1F
        g6 = (g >> 2) & 0x3F
        b5 = (b >> 3) & 0x1F
        px.append((r5 << 11) | (g6 << 5) | b5)
    return img.size, px


def main():
    src = find_logo()
    if src is None:
        print("在 utils/ 下没找到 logo.png / logo.jpg / logo.bmp")
        print("请把你的 LOGO 图片放进去（任意尺寸，会自动缩放到 %dx%d）"
              % (SCREEN_W, SCREEN_H))
        sys.exit(1)

    img = Image.open(src)
    (w, h), pixels = to_rgb565(img)
    print("源图片: %s  %s  ->  %dx%d RGB565" % (os.path.basename(src), img.size, w, h))

    lines = []
    lines.append("// LogoImage.h - 由 utils/logo_converter.py 自动生成，请勿手工编辑")
    lines.append("// 源图片: %s  尺寸: %dx%d  格式: RGB565 大端友好" % (os.path.basename(src), w, h))
    lines.append("#ifndef LOGO_IMAGE_H")
    lines.append("#define LOGO_IMAGE_H")
    lines.append("")
    lines.append("#include <Arduino.h>")
    lines.append("")
    lines.append("#define LOGO_W %d" % w)
    lines.append("#define LOGO_H %d" % h)
    lines.append("")
    lines.append("// 存储顺序与屏幕扫描一致，直接逐行推送即可")
    lines.append("static const uint16_t LOGO_BITMAP[%d] PROGMEM = {" % (w * h))
    for i in range(0, len(pixels), 12):
        row = ", ".join("0x%04X" % v for v in pixels[i:i + 12])
        lines.append("    " + row + ",")
    lines.append("};")
    lines.append("")
    lines.append("#endif  // LOGO_IMAGE_H")

    os.makedirs(os.path.dirname(OUT_PATH), exist_ok=True)
    with open(OUT_PATH, "w", encoding="utf-8", newline="\n") as f:
        f.write("\n".join(lines))

    print("已生成: include/LogoImage.h (%d 像素, 约 %d KB flash)"
          % (len(pixels), len(pixels) * 2 // 1024))
    print("下一步: pio run -e esp32c3 -t upload --upload-port COM30")


if __name__ == "__main__":
    main()
