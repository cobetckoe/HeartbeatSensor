# -*- coding: utf-8 -*-
"""
make_logo.py - 坎川工业 LOGO 设计脚本 (v2: 环形龙齿轮 + 晋字)

设计说明 (通用企业标识, 不含任何雷达元素):
  - 外圈: 圆环形齿轮 (16 个尖齿), 齿在外缘
          环身即龙身 -> 自顶部龙颈(最宽)沿环顺时针渐变收细, 绕回顶部收为尾尖
          龙首自环顶向外昂起 (张口 / 鹿角 / 双鬃), 形成首尾相衔的盘龙齿轮环
  - 内圈: 山西 "晋" 字 (三晋 / 龙城元素, 居中)
  - 下方: 坎川工业
  全部由直线/圆弧多边形构成, 便于移植为单片机 GFX 图元

输出:
  utils/logo_gear_dragon.png      800x800 彩色主图
  utils/logo_gear_dragon_mono.png 800x800 单色 (印刷 / 矢量化)
  utils/logo_gear_dragon_icon.png 256x256 图标 (无文字)
  utils/logo_160x128.png          160x128 屏幕像素预览
  utils/logo.png                  160x128 供 logo_converter.py 转固件点阵

用法: python utils/make_logo.py
依赖: pip install Pillow
"""

import math
import os
from PIL import Image, ImageDraw, ImageFont

HERE = os.path.dirname(os.path.abspath(__file__))

FONT_CANDIDATES = [
    r"C:\Windows\Fonts\simhei.ttf",
    r"C:\Windows\Fonts\msyhbd.ttc",
    r"C:\Windows\Fonts\msyh.ttc",
]

# ---------------- 配色 ----------------
PAL_MAIN = {
    "bg":    (13, 19, 17),      # 深墨绿黑
    "ring":  (220, 182, 106),   # 军工金 (齿轮环 / 龙)
    "jin":   (198, 60, 48),     # 朱红 (晋字)
    "text":  (232, 226, 208),   # 米白 (公司名)
}
PAL_MONO = {
    "bg":    (255, 255, 255),
    "ring":  (18, 18, 18),
    "jin":   (18, 18, 18),
    "text":  (18, 18, 18),
}
PAL_DEV = {
    "bg":    (6, 14, 9),
    "ring":  (216, 196, 136),
    "jin":   (224, 72, 64),
    "text":  (232, 226, 208),
}


def load_font(size):
    for p in FONT_CANDIDATES:
        if os.path.isfile(p):
            try:
                return ImageFont.truetype(p, size)
            except Exception:
                continue
    return ImageFont.load_default()


def polar(cx, cy, r, th):
    return (cx + r * math.cos(th), cy + r * math.sin(th))


def fit_font(text, target_h, max_w=None):
    """返回字号, 使 text 实际渲染高度接近 target_h。"""
    probe = ImageDraw.Draw(Image.new("RGB", (8, 8)))
    f = load_font(100)
    bb = probe.textbbox((0, 0), text, font=f)
    h = max(1, bb[3] - bb[1])
    size = max(6, int(round(100.0 * target_h / h)))
    if max_w:
        f2 = load_font(size)
        bb2 = probe.textbbox((0, 0), text, font=f2)
        w = max(1, bb2[2] - bb2[0])
        if w > max_w:
            size = max(6, int(size * max_w / w))
    return load_font(size)


# ---------------------------------------------------------------------------
#  环形龙齿轮 (首尾相衔: 环身沿环渐细, 缺口处昂首)
# ---------------------------------------------------------------------------
def gear_dragon_shape(cx, cy, R, n_teeth=16, half_gap=25.0, w_neck=0.27,
                       r_mid=0.855, tooth_h=0.115, samples=1440,
                       r_min_frac=0.004):
    """返回 (闭合多边形, 端口线起点, 起点外法向, 尖端)。
    环自缺口右侧顺时针绕一周, 宽度从颈部渐变到尖端。"""
    a0 = math.radians(-90.0 + half_gap)      # 缺口右侧 = 颈部
    sweep = 2.0 * math.pi - 2.0 * math.radians(half_gap)
    rm = r_mid * R
    pos_len = n_teeth / 16.0                 # 齿相对密度

    def halfw(t):
        return R * (w_neck * (1.0 - t) ** 0.75 + r_min_frac)

    def tooth_off(th, h, t):
        # 齿落在环身等分位上, 齿高随环宽收窄
        u = t * n_teeth
        dth = (u - round(u)) * (sweep / n_teeth)
        half = (sweep / n_teeth) * 0.32
        a = abs(dth)
        if a >= half:
            return 0.0
        return h * (1.0 - a / half)

    outer, inner = [], []
    for i in range(samples + 1):
        t = i / samples
        th = a0 + t * sweep
        w = halfw(t)
        # 靠近尾尖处齿高淡出, 避免末端凸出一根齿刺
        fade = 1.0 if t < 0.82 else max(0.0, (1.0 - t) / 0.18)
        h = tooth_h * R * (0.35 + 0.65 * min(1.0, w / (R * w_neck))) * fade
        # 尾段逐渐向环心内收 (龙尾内卷)
        rc = rm - 0.055 * R * (t ** 4)
        outer.append(polar(cx, cy, rc + w * 0.5 + tooth_off(th, h, t), th))
    for i in range(samples, -1, -1):
        t = i / samples
        th = a0 + t * sweep
        rc = rm - 0.055 * R * (t ** 4)
        inner.append(polar(cx, cy, rc - halfw(t) * 0.5, th))

    t0 = math.radians(-90.0 + half_gap)
    neck = polar(cx, cy, rm, t0)
    outw = (math.cos(t0), math.sin(t0))       # 颈部外法向(径向向外)
    tip = polar(cx, cy, rm, math.radians(-90.0 - half_gap))
    return outer + inner, neck, outw, tip


def tapered_poly(spine, widths):
    """由脊线 + 逐点半宽生成平滑闭合多边形 (龙角/鬃/尾等)。"""
    n = len(spine)
    left, right = [], []
    for i, ((x, y), w) in enumerate(zip(spine, widths)):
        a = spine[max(i - 1, 0)]
        b = spine[min(i + 1, n - 1)]
        dx, dy = b[0] - a[0], b[1] - a[1]
        L = math.hypot(dx, dy) or 1.0
        nx, ny = -dy / L, dx / L
        left.append((x + nx * w, y + ny * w))
        right.append((x - nx * w, y - ny * w))
    return left + right[::-1]


def dragon_head(px, py, tx, ty, S, r_eye_frac=0.030):
    """龙首 (局部: x 沿面朝方向, y 沿背部)。

    返回 (实心块列表, 眼睛圆心, 眼睛半径, 鼻孔圆心)。
    全部为凸多边形/单缺口多边形, 填色即可, 不做描边 (描边会把角鬃切碎)。
    """
    nx, ny = -ty, tx

    def L2W(lx, ly):
        return (px + lx * S * tx + ly * S * nx,
                py + lx * S * ty + ly * S * ny)

    # 头: 厚颅 + 长吻 + 深张的口 (单一凹多边形, 填充无描边)
    #   局部 +x = 面朝方向(吻尖), +y = 背侧(背离环心)
    head2d = [
        (-0.195, 0.145),                        # 颈背
        (-0.075, 0.185),                        # 颅后
        (0.030, 0.205),                         # 眉弓
        (0.150, 0.180),                         # 额
        (0.290, 0.135),                         # 吻背
        (0.410, 0.085),                         # 上颚
        (0.495, 0.070),                         # 吻尖
        (0.325, 0.010),                         # 口内 (深 V 顶点)
        (0.440, -0.050),                        # 下颚尖
        (0.330, -0.105),                        # 下颚
        (0.170, -0.145),                        # 颐
        (0.020, -0.170),                        # 喉
        (-0.195, -0.120),                       # 颈下
    ]
    # 双角: 主角自颅顶向后上方长掠, 副角略短
    horn1 = tapered_poly([(0.030, 0.200), (-0.085, 0.305), (-0.200, 0.380)],
                         [0.066, 0.040, 0.011])
    horn2 = tapered_poly([(-0.075, 0.172), (-0.185, 0.248), (-0.285, 0.295)],
                         [0.044, 0.026, 0.008])
    # 颔须 (自下颌向后下方飘出)
    mane2 = tapered_poly([(0.150, -0.135), (0.020, -0.215), (-0.115, -0.265)],
                         [0.038, 0.024, 0.007])
    parts = [horn1, horn2, mane2, head2d]
    polys = [[L2W(*pt) for pt in p2] for p2 in parts]
    return polys, L2W(0.155, 0.085), S * r_eye_frac, L2W(0.455, 0.080)

# ---------------------------------------------------------------------------
#  整体绘制
# ---------------------------------------------------------------------------
def draw_all(d, size, pal, with_text=True, R_ratio=0.330, n_teeth=16,
             head_scale=1.30, name_ratio=0.195, jin_target=None,
             half_gap=25.0, jitter=-90.0):
    R = int(size * R_ratio)
    cx = size // 2
    cy = int(size * 0.472) if with_text else size // 2
    rm = 0.855 * R

    poly, neck, outw, tip = gear_dragon_shape(
        cx, cy, R, n_teeth=n_teeth, half_gap=half_gap)

    # 1) 环形龙齿轮 (环身 + 齿)
    d.polygon(poly, fill=pal["ring"])

    # 2) 龙首: 自颈部昂起, 越过缺口 -> 与另一端的尾尖首尾相衔
    t0 = math.radians(-90.0 + half_gap)
    tang = (-math.sin(t0), math.cos(t0))          # 环身逆角度方向
    hx = cx + rm * math.cos(t0)
    hy = cy + rm * math.sin(t0)
    S = R * head_scale
    polys, eye, er, nostril = dragon_head(hx, hy, -tang[0], -tang[1], S)
    # 龙首覆于环身之上 (纯填充, 无描边)
    for p in polys:
        d.polygon(p, fill=pal["ring"])
    r_eye = max(1, int(er))
    d.ellipse([eye[0] - r_eye, eye[1] - r_eye, eye[0] + r_eye, eye[1] + r_eye],
              fill=pal["bg"])
    rn = max(1, int(er * 0.55))
    d.ellipse([nostril[0] - rn, nostril[1] - rn,
               nostril[0] + rn, nostril[1] + rn], fill=pal["bg"])

    # 3) 中心 晋 字
    inner_r = rm - 0.5 * R * (0.27 + 0.004)       # 颈部处内孔半径(最小)
    tgt = jin_target if jin_target else inner_r * 1.30
    f_jin = fit_font("晋", tgt, max_w=inner_r * 1.56)
    bb = d.textbbox((0, 0), "晋", font=f_jin)
    jw, jh = bb[2] - bb[0], bb[3] - bb[1]
    d.text((cx - jw / 2 - bb[0], cy - jh / 2 - bb[1]), "晋",
           font=f_jin, fill=pal["jin"])

    # 4) 公司名
    if with_text:
        f_nm = fit_font("坎川工业", R * name_ratio, max_w=size * 0.82)
        bb2 = d.textbbox((0, 0), "坎川工业", font=f_nm)
        nw = bb2[2] - bb2[0]
        ty2 = cy + int(1.17 * R)
        d.text((cx - nw / 2 - bb2[0], ty2), "坎川工业",
               font=f_nm, fill=pal["text"])


def save(size, out_path, pal, **kw):
    img = Image.new("RGB", (size, size), pal["bg"])
    draw_all(ImageDraw.Draw(img), size, pal, **kw)
    img.save(out_path)
    print("saved:", out_path)
    return img


def main():
    save(800, os.path.join(HERE, "logo_gear_dragon.png"), PAL_MAIN)
    save(800, os.path.join(HERE, "logo_gear_dragon_mono.png"), PAL_MONO)
    save(256, os.path.join(HERE, "logo_gear_dragon_icon.png"), PAL_MAIN,
         with_text=False, R_ratio=0.40)

    # 160x128 屏幕预览 (仅图形, 居中)
    scr = Image.new("RGB", (160, 128), PAL_DEV["bg"])
    tmp = Image.new("RGB", (128, 128), PAL_DEV["bg"])
    draw_all(ImageDraw.Draw(tmp), 128, PAL_DEV, with_text=False, R_ratio=0.40)
    scr.paste(tmp, (16, 0))
    p = os.path.join(HERE, "logo_160x128.png")
    scr.save(p)
    print("saved:", p)
    p2 = os.path.join(HERE, "logo.png")
    scr.save(p2)
    print("saved:", p2, "(logo_converter.py 输入)")


if __name__ == "__main__":
    main()
